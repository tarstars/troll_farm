/// Referee v1.0.5 end-of-game rule (engine/Board.java `hasStalled`, ported verbatim).
/// Call AFTER `step` each turn with a persistent counter (start 0). While plants exist
/// the grace counter is recomputed from units standing on plants (walk-home time + 6)
/// and the game continues. With no plants: counter ticks down (<=0 ends); a player is
/// "stuck" unless a unit carries a non-iron item or their bank holds any fruit; both
/// stuck ends; a stuck player who is strictly losing ends (mercy rule).
pub fn has_stalled(game: &GameState, turns_until_end: &mut i32) -> bool {
    if !game.plants.is_empty() {
        *turns_until_end = 0;
        for u in &game.units {
            let pos = u.pos();
            if plant_at_pos(&game.plants, pos).is_none() {
                continue;
            }
            let shack = game.shacks[u.player as usize];
            let dist = bfs_dist_to(game, pos, shack);
            *turns_until_end = (*turns_until_end).max(dist / u.ms.max(1) + 6);
        }
        return false;
    }
    *turns_until_end -= 1;
    if *turns_until_end <= 0 {
        return true;
    }
    let mut stuck = [true, true];
    for u in &game.units {
        let non_iron: i32 = u.carry[0] + u.carry[1] + u.carry[2] + u.carry[3] + u.carry[5];
        if non_iron > 0 {
            stuck[u.player as usize] = false;
        }
    }
    for p in 0..2 {
        if game.inventories[p][0..4].iter().any(|&c| c > 0) {
            stuck[p] = false;
        }
    }
    if stuck[0] && stuck[1] {
        return true;
    }
    if stuck[0] && game.scores[0] < game.scores[1] {
        return true;
    }
    if stuck[1] && game.scores[1] < game.scores[0] {
        return true;
    }
    false
}

/// BFS distance from `from` to the walkable cells adjacent to `target` (the shack cell
/// itself is unwalkable; matches the referee's getDistances-to-shack semantics of
/// "distance to the shack's cell" — seed the shack cell at 0 and expand over walkable,
/// exactly like the referee seeds unwalkable sources).
fn bfs_dist_to(game: &GameState, from: Cell, target: Cell) -> i32 {
    use std::collections::{HashMap, VecDeque};
    let mut dist: HashMap<Cell, i32> = HashMap::new();
    let mut q: VecDeque<Cell> = VecDeque::new();
    dist.insert(target, 0);
    q.push_back(target);
    while let Some((x, y)) = q.pop_front() {
        let d = dist[&(x, y)];
        if (x, y) == from {
            return d;
        }
        for n in [(x, y + 1), (x + 1, y), (x, y - 1), (x - 1, y)] {
            if game.walkable.contains(&n) && !dist.contains_key(&n) {
                dist.insert(n, d + 1);
                q.push_back(n);
            }
        }
    }
    9999
}
